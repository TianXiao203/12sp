# CI 构建开关（push 触发）
#
#   KSU=on|off            on=集成 ReSukiSU（默认）
#   FRAGMENT=all|docker|ns|userns|ipc|cgroup|none
#                         docker = 【最终交付配置】：整份碎片 - CGROUP_DEVICE/PIDS
#                         all    = 原样（含 ABI 危险项，只用于对照实验）
#                         其余    = bisect 分组，仅排查用
#   DEBUG_INFO=on|off     默认 off（16GB runner 上 FULL LTO 会 OOM 的对策）
#
# bisect 结论回顾（详见 configs/docker-cgroup.fragment 开头与
# scripts/patch-abi-safe.sh）：
#   control（设备原版配置）        → ABI 全绿 ⇒ 流水线/编译器/基线都对
#   D2（none）                     → 全绿     ⇒ PID_NS=y 安全、DEBUG_INFO=off 安全
#   E2（none+KSU）                 → 全绿     ⇒ ReSukiSU 安全
#   G（只开 USER_NS）              → 全绿     ⇒ USER_NS 安全（5.10 不 gate 结构体成员）
#   B（只开 CGROUP_DEVICE+PIDS）   → 红 295   ⇒ struct css_set 定长数组变长
#   H（POSIX_MQUEUE+IPC_NS+SYSCTL) → 红 725   ⇒ struct user_struct 多 mq_bytes
#   F2（整个 ns 组）               → 红 725，且与 H 的失配集合逐个 CRC 完全相同
#   ⇒ 攻击面只剩两处：CSS_SET（放弃两个 cgroup 控制器）
#                     及 USER_STRUCT（用 ANDROID_KABI_USE 挪进 KABI 保留槽，已修）
#
# --- 最终交付：ABI 安全的 Docker 配置 ---
KSU=on
FRAGMENT=docker
DEBUG_INFO=off
